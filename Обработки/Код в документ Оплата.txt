&НаКлиенте
Процедура ПечатьАктНовый(Документ, MSWord)
	
	Для А=1 По 2 Цикл

	  	ОбщаяСуммаК = Объект.СводнаяТаблица.Итог("Факт");
		Замена = Документ.Content.Find; 
		Замена.Execute("<Всего>", Ложь, Истина, Ложь, , , Истина, , Ложь, ОбщаяСуммаК);
	КонецЦикла;

	Замена = Документ.Content.Find; 
	Замена.Execute("<Принял>", Ложь, Истина, Ложь, , , Истина, , Ложь, Формат(Объект.Преподаватель));
	
	       
ТаблТек = Документ.Tables(1);
		   
    	     Сч = 1;
    					
    	Для каждого СтрокаСумм Из Объект.СводнаяТаблица Цикл
    		Дата = Формат(СтрокаСумм.Дата,"ДФ=dd.MM.yyyy; ДЛФ=D; ДП=-");
			Группа = Строка(СтрокаСумм.Группа);
			НаименованиеДисциплина = Строка(СтрокаСумм.Дисциплина);
			ВидЗанятия = Строка(СтрокаСумм.ВидЗанятия);
			КоличествоЧасов = СтрокаСумм.Факт;
    		ТаблТек.Cell(Сч+1,1).Range.Text = Дата;
    		ТаблТек.Cell(Сч+1,2).Range.Text = Группа;
			ТаблТек.Cell(Сч+1,3).Range.Text = НаименованиеДисциплина;
			ТаблТек.Cell(Сч+1,4).Range.Text = ВидЗанятия;
			ТаблТек.Cell(Сч+1,5).Range.Text = КоличествоЧасов; 	
			ТаблТек.Rows.Add();
			
			Сч = Сч + 1;
		КонецЦикла;
		    	
	MSWord.Visible = 1; 
	MSWord.Activate();
	Объект.АктРаспечатан=Истина;

КонецПроцедуры
